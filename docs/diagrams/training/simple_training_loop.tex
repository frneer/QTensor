\begin{tikzpicture}[
    block/.style={rectangle, draw, fill=white, text width=4.5em, text centered, rounded corners, minimum height=2em},
    arrow/.style={-latex, thick}
]
    \tikzset{node distance = 1cm and 2cm}
    % Nodes
    \node (x_input) at (0, 0) [block] {$x$};
    \node (w_input) [below=of x_input] [block] {$w$};
    \node (forward_pass) [right=of w_input] [block] {Forward Pass};
    \node (loss) [below=of forward_pass] [block] {Loss};
    \node (y_input) [below=of loss] [block] {$y$};
    \node (gradient) [left=of loss] [block] {Backward Pass};

    % % Arrows
    \draw [arrow] (x_input) -| (forward_pass);
    \draw [arrow] (w_input) -- (forward_pass); % Connect W to forward pass
    \draw [arrow] (y_input) -- (loss);
    \draw [arrow] (forward_pass) -- node[below=of forward_pass, yshift = 1.5cm, xshift = -0.2cm] {$\hat{y}$} (loss);
    \draw [arrow] (loss) -- (gradient);
    \draw [arrow] (gradient) --  (w_input); % Gradient updates W

\end{tikzpicture}
