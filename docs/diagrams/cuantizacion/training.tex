\begin{tikzpicture}[
    block/.style={rectangle, draw, fill=white, text width=6em, text centered, rounded corners, minimum height=3em, minimum width=6em, font=\footnotesize},
    arrow/.style={-latex, thick}
]
    \tikzset{node distance = 1.5cm and 1.5cm}
    % Nodes
    \node (w_input) at (0,0) [block] {Actualización de parámetros};
    \node (forward_pass) [right=of w_input] [block] {Inferencia};
    \node (loss) [below=of forward_pass] [block] {Calculo del costo};
    \node (gradient) [left=of loss] [block] {Cálculo del gradiente};
    \node (x_input) [above=of forward_pass, yshift=-0.5cm] [block] {$x$};
    \node (y_input) [below=of loss, yshift=0.5cm] [block] {$y$};

    % % Arrows
    \draw [arrow] (x_input) -- (forward_pass);
    \draw [arrow] (w_input) -- node[midway, above] {$\theta$} (forward_pass); % Connect W to forward pass
    \draw [arrow] (y_input) -- (loss);
    \draw [arrow] (forward_pass) -- node[midway, right] {$\hat{y}$} (loss);
    \draw [arrow] (loss) -- node[midway, above] {$\mathcal{L}$} (gradient);
    \draw [arrow] (gradient) -- node[midway, left] {$\frac{\partial \mathcal{L}}{\partial \theta}$} (w_input); % Gradient updates W

\end{tikzpicture}
