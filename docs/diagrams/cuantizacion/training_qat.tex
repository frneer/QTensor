\begin{tikzpicture}[
    block/.style={rectangle, draw, fill=white, text width=4em, text centered, rounded corners, minimum height=2em},
    rectangular_block/.style={rectangle, draw, fill=white, text width=4em, text centered, minimum height=2em},
    arrow/.style={-latex, thick}

    \tikzset{node distance = 1cm and 2cm}
    % Nodes
    \node (x_input) at (0, 0) [rectangular_block] {$x$};

    \node (loss) [below=of forward_pass] [block] {$\mathcal{L}$};

    \draw [arrow] (w_input) -- (quantizer); % Connect W to quantizer

    \draw [arrow] (y_input) -- (loss);
    \draw [arrow] (forward_pass) -- node[below=of forward_pass, yshift = 1.5cm, xshift = -0.2cm] {$\hat{y}$} (loss);
    \draw [arrow] (loss) -- (gradient);
    % \draw [arrow] (gradient_q) -- (gradient);
    \draw [arrow] (gradient) -- (w_input); % Gradient updates W
